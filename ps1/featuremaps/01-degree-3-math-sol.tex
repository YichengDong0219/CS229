\begin{answer}
    \subsubsection*{1) J($\theta$) with linear Regression on the dataset $\{\hat{x}^{(i)}, y^{(i)}\}_{i=1}^n:$}

    Let $\hat{x} \in \mathbb{R}^d$ = $\phi(x)$,
        $\hat{x}^{(i)} = \phi(x^{(i)})$,
        $\phi(x) = \begin{bmatrix} 1 \\ x \\ x^2 \\ x^3 \end{bmatrix}$,
        $\hat{X} = \begin{bmatrix} \phi(x^{(1)})^T \\ \vdots \\ \phi(x^{(n)})^T \end{bmatrix}$
    
    \setcounter{equation}{0}
    \begin{align}
        J(\theta) &= \frac{1}{2} ||\hat{X}\theta - y||_2^2 \\
        &= \frac{1}{2} \sum_{i=1}^n (\theta^T\hat{x}^{(i)} - y^{(i)})^2 \\
        &= \frac{1}{2} \sum_{i=1}^n (\theta^T\phi(x^{(i)}) - y^{(i)})^2
    \end{align}
    
    \subsubsection*{2) Update rule of Batch GD on dataset $\{ \hat{x}^{(i)}, y^{(i)}\}_{i=1}^n:$}
    (1) $\theta^{t+1} = \theta^t - \alpha \nabla J(\theta)$ \\
    Gradient for regular LMS: $\nabla_\theta J(\theta) = X^T(X\theta - y)$ \\
    $\to$ for our case $\{ \hat{x}^{(i)}, y^{(i)}\}_{i=1}^n:$
    % --- SCRIBBELED OUT SECTION ---
    $\nabla J(\theta) = \hat{X}^T(\hat{X}\theta - y)$ \\
    
    Inserting into the update rule: \\
    $(2)^* \quad \theta^{t+1} = \theta^t - \alpha(\hat{X}^T(\hat{X}\theta-y))$
