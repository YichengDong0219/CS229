\begin{answer}
    
    \textbf{1) Latent Variables}

    With the E-step, we want to estimate the each cluster's responsibility or more simply the cluster assignments $\{z^{(1)}, \ldots, z^{(n)}\}$ for the $n$ unlabeled examples $\{x^{(1)}, \ldots, x^{(n)}\}$. So the E-step is about the $n$ unlabeled examples.

    \textbf{2) E-step derivation}

    We need to compute $Q_i^{(t)}(z^{(i)}=j) = p(z^{(i)}=j | x^{(i)}; \theta^{(t)})$, the probability of the $i$-th unlabeled data point $x^{(i)}$ belonging to cluster $j$ with the current $\theta^{(t)} = (\phi^{(t)}, \mu^{(t)}, \Sigma^{(t)})$.

    Bayes rule can be used to rewrite that probability:

    $$p(z^{(i)}=j | x^{(i)}; \theta^{(t)}) = \frac{p(x^{(i)}|z^{(i)}=j; \theta^{(t)}) p(z^{(i)}=j; \theta^{(t)})}{p(x^{(i)}; \theta^{(t)})}$$

    \begin{itemize}
        \item $p(z^{(i)}=j; \theta^{(t)})$ $\rightarrow$ That is the probability of cluster $j$, so: $p(z^{(i)}=j; \theta^{(t)}) = \phi_j^{(t)}$
        \item $p(x^{(i)}|z^{(i)}=j; \theta^{(t)})$ $\rightarrow$ That is the Gaussian PDF for cluster $j$, so: $p(x^{(i)}|z^{(i)}=j; \theta^{(t)}) = \mathcal{N}(x^{(i)}; \mu_j^{(t)}, \Sigma_j^{(t)})$
        \item $p(x^{(i)}; \theta^{(t)}) = \sum_{l=1}^{k} p(x^{(i)}, z^{(i)}=l; \theta^{(t)}) = \sum_{l=1}^{k} \phi_l^{(t)} \mathcal{N}(x^{(i)}; \mu_l^{(t)}, \Sigma_l^{(t)})$
    \end{itemize}

    That gives us:

    $$Q_i^{(t)}(z^{(i)}=j) := \frac{\phi_j^{(t)} \mathcal{N}(x^{(i)}; \mu_j^{(t)}, \Sigma_j^{(t)})}{\sum_{l=1}^{k} \phi_l^{(t)} \mathcal{N}(x^{(i)}; \mu_l^{(t)}, \Sigma_l^{(t)})}$$
\end{answer}
